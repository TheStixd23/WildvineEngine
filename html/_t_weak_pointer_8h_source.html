59 TWeakPointer(const TSharedPointer<T>& sharedPtr)
60 : ptr(sharedPtr.ptr), refCount(sharedPtr.refCount) {
61 }
62
68 TSharedPointer<T>
69 lock() const {
70 if (refCount && *refCount > 0) {
71 return TSharedPointer<T>(ptr, refCount);
72 }
73 return TSharedPointer<T>();
74 }
75
76 // Reset
77 void
78 reset() {
79 ptr = nullptr;
80 refCount = nullptr;
81 }
