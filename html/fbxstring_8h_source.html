500inline void FbxRemoveChar(FbxString& pString, char pToRemove)
501{
502 int lPos = pString.ReverseFind(pToRemove);
503 while( lPos >= 0 )
504 {
505 pString = pString.Left(lPos) + pString.Mid(lPos + 1);
506 lPos = pString.ReverseFind(pToRemove);
507 }
508}
