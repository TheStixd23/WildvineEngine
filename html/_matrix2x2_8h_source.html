137 Matrix2x2 inverse() const {
138 float det = determinant();
139 if (det == 0) {
140 // Handle non-invertible matrix gracefully.
141 return Matrix2x2();
142 }
143 float invDet = 1.0f / det;
144 return Matrix2x2(
145 m[1][1] * invDet, -m[0][1] * invDet,
